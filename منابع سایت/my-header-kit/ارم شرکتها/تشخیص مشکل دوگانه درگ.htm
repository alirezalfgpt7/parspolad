javascript
// --- جایگزین تابع initializeDragAndDrop + handleDrop در WebDesignTool ---
initializeDragAndDrop() {
    if (!this.designCanvas) return;

    // اگر قبلاً مقداردهی شده، از ثبت مجدد جلوگیری کن
    if (this._dragAndDropInitialized) return;
    this._dragAndDropInitialized = true;

    // guard برای جلوگیری از اجرای همزمان/دوگانهٔ drop
    this._dropProcessing = false;
    this._lastDropTime = 0; // timestamp آخرین drop

    // dragover (اجازهٔ drop)
    this.designCanvas.addEventListener('dragover', (e) => {
        e.preventDefault();
        this.handleDragOver(e);
    });

    // drop - این هندلر با guard محافظت شده است
    this.designCanvas.addEventListener('drop', (e) => {
        e.preventDefault();

        // محافظت سریع: اگر اخیراً drop پردازش شده، از ادامه جلوگیری کن
        const now = Date.now();
        if (this._dropProcessing) {
            console.warn('Drop ignored because another drop is processing.');
            return;
        }
        if (this._lastDropTime && (now - this._lastDropTime) < 200) {
            // اگر کمتر از 200ms از drop قبلی گذشته بود احتمال اجرای دوگانه وجود داره — جلوگیری کن
            console.warn('Drop ignored due to rapid repeat (debounce).');
            return;
        }

        try {
            this._dropProcessing = true;
            this._lastDropTime = now;

            // اجرای واقعی drop را به تابع جداگانه می‌سپاریم
            this.handleDrop(e);
        } finally {
            // کمی تاخیر برای اطمینان از اینکه دیگر اجراها نریزند
            setTimeout(() => {
                this._dropProcessing = false;
            }, 250); // 250ms قفل بعد از drop (قابل تنظیم)
        }
    });

    // dragleave
    this.designCanvas.addEventListener('dragleave', (e) => {
        this.handleDragLeave(e);
    });
}

// تابع handleDrop را جدا نگه دار تا قابل unit-test و لاگ باشد
handleDrop(e) {
    // این تابع فرض می‌کند e.preventDefault() قبلاً انجام شده است
    try {
        const elementType = e.dataTransfer && e.dataTransfer.getData && e.dataTransfer.getData('text/plain');
        if (!elementType) {
            console.warn('No element type in drop data');
            return;
        }

        const targetElement = this.getElementUnderCursor(e);
        const canvasRect = this.designCanvas ? this.designCanvas.getBoundingClientRect() : { left: 0, top: 0 };
        const x = e.clientX - canvasRect.left;
        const y = e.clientY - canvasRect.top;

        console.log('Drop event handled (single-run guard active):', { elementType, targetElementId: targetElement ? targetElement.id : null, x, y });

        // اگر parent امکان nesting دارد، از مسیر createNestedElement استفاده کن
        if (targetElement && this.elementManager && this.elementManager.canElementNest(targetElement.type)) {
            this.elementManager.createNestedElement(elementType, targetElement, e.clientX, e.clientY);
        } else {
            this.elementManager.createElement(elementType, x, y, null);
        }

        this.hideDropZones();
    } catch (err) {
        console.error('Drop handler error (inner):', err);
        this.showNotification('Error dropping element: ' + (err.message || ''), 'error');
    }
}

// --- جایگزین تابع createNestedElement در ElementManager ---
createNestedElement(type, parentElement, offsetX, offsetY) {
  if (!type) {
    console.warn('createNestedElement called without type');
    return null;
  }

  if (!parentElement || !parentElement.id) {
      console.warn('Invalid parent element for nesting, falling back to root creation');
      // اگر parent نامعتبر است از createElement معمولی استفاده کن
      return this.createElement(type, offsetX, offsetY, null);
  }

  const parentDiv = document.getElementById(parentElement.id);
  if (!parentDiv) {
      console.warn('Parent element DOM not found, creating as root element');
      return this.createElement(type, offsetX, offsetY, null);
  }

  const parentRect = parentDiv.getBoundingClientRect();
  const relativeX = Math.max(0, offsetX - parentRect.left);
  const relativeY = Math.max(0, offsetY - parentRect.top);

  // بررسی تکراری بودن قبل از ایجاد المان
  const existing = this.isDuplicateElement(type, relativeX, relativeY, parentElement.id);
  if (existing) {
      console.log('Duplicate element prevented (createNestedElement):', type, 'near', relativeX, relativeY, 'parent', parentElement.id);
      return existing;
  }

  // یکبار فراخوانی createElement با parentId مشخص
  return this.createElement(type, relativeX, relativeY, parentElement.id);
}

چرا: اطمینان از اینکه هر المان تنها یک‌بار به دیتاست اضافه شود و رندر نیز فقط یکبار اجرا گردد (حفاظت‌های داخلی). همچنین لاگ جهت دیباگ.
کد جایگزین کامل تابع:

javascript
// --- جایگزین تابع createElement در ElementManager ---
createElement(type, x = 0, y = 0, parentId = null) {
    if (!type) {
        console.warn('createElement called without type');
        return null;
    }

    const elementId = `elem-${Date.now()}-${++this.elementCounter}`;
    const elementConfig = this.getElementConfig(type) || {};

    const element = {
        id: elementId,
        type: type,
        x: typeof x === "number" ? x : 0,
        y: typeof y === "number" ? y : 0,
        parentId: parentId,
        properties: this.getDefaultProperties(type),
        styles: this.getDefaultStyles(type),
        animations: [],
        children: [],
        zIndex: this.getNextZIndex(),
        responsive: { mobile: {}, tablet: {}, desktop: {} },
        _configName: elementConfig.name || type,
    };

    // جلوگیری از اضافه شدن چندباره (double insert)
    let elementAdded = false;

    if (parentId) {
        const parentElement = this.findElementById(parentId);
        if (parentElement) {
            parentElement.children = parentElement.children || [];
            // امنیت: مطمئن شویم این آبجکت قبلاً اضافه نشده
            if (!parentElement.children.some(c => c === element || c.id === element.id)) {
                parentElement.children.push(element);
                elementAdded = true;
                console.log('Element added to parent:', parentId, element.id);
            } else {
                console.warn('Attempt to add element to parent twice prevented', element.id);
            }
        } else {
            console.warn('ParentId provided but parent not found, adding to root instead', parentId);
        }
    }

    if (!elementAdded) {
        // مطمئن شویم که عنصر با همان id در root موجود نیست
        if (!this.designTool.elements.some(el => el.id === element.id)) {
            this.designTool.elements.push(element);
            elementAdded = true;
            console.log('Element added to root', element.id);
        } else {
            console.warn('Element already exists in root, preventing duplicate', element.id);
        }
    }

    if (elementAdded) {
        // فقط اگر DOM هنوز این المان را نداشت، آن را رندر کن
        try {
            this.renderElement(element);
            this.designTool.setSelectedElement(element);

            if (this.designTool.propertyManager && typeof this.designTool.propertyManager.updatePropertyPanel === "function") {
                this.designTool.propertyManager.updatePropertyPanel();
            }

            if (typeof this.designTool.saveState === "function") {
                this.designTool.saveState();
            }
        } catch (err) {
            console.error('Error during createElement render stage:', err);
        }
    }

    return element;
}

// اگر تابع attachElementDragEvents وجود دارد، آن را به این شکل ایمن کن:
attachElementDragEvents() {
    if (this._elementDragEventsAttached) return;
    this._elementDragEventsAttached = true;

    const elementItems = document.querySelectorAll('.element-item');
    elementItems.forEach(item => {
        // ابتدا remove کن تا دبل لیسنر نداشته باشیم (در صورت DOM re-render)
        item.removeEventListener('dragstart', this.handleDragStart);
        item.addEventListener('dragstart', this.handleDragStart);
    });
}

منبع/نمونه: بخشی از attach logic را در فایل‌ها دیدم؛ این تغییر مانع دوبار ثبت شدن listener در صورت فراخوانی مجدد تابع می‌شود.

هم‌چنین (رفع مشکل نوار عملیات که روی خود المان ظاهر می‌شود)

تو توضیحات گفته‌ای: «نوار حذف/duplicate/up-down روی خود المان هستند و حتی در پیش‌نمایش هم دیده می‌شوند». علت معمول: شما generateElementActions را مستقیماً داخل elementDiv.innerHTML می‌ریزید (یعنی خود اکشن‌ها جزِ HTML المان می‌شوند). راه بهتر: اکشن‌ها را در یک عنصر overlay جداگانه در DOM (مثلاً .element-overlay) قرار دهید و آن را با position:absolute در بالای المان قرار بدهید — یا حداقل هنگام ایجاد preview از رندر کردن اکشن‌ها در داخل iframe پیش‌نمایش جلوگیری کنی.
